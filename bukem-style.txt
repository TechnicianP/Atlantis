_$amen_chopped:
  s("breaks/2")
    .n(1)
    .fit()
    // chop into 16 but play 8th-note long segments
    .scrub(irand(16).div(16).seg(2))
    // 10% of the time repeat 2 or 4 times
    .almostNever(ply("2 | 4"))
    .gain(0.9)
    .orbit(2)
