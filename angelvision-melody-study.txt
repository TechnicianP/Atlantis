setCpm(70/4)

$kick: s("bd:4").beat("0,1,3,11", 16).duckorbit(5).duckdepth(1).duckattack(.4).dec("<.05 1@4>/4")
$kick2: s("bd:3").beat(14, 16).speed(rand.range(.5,.8)).delay(.5)

$snap: s("cp:5").beat("2,4,7,15", 16).speed(".8 .7 .6 <.5 .4>").degradeBy(.3).room(.6).gain(.3)
$rim: s("cp:0").beat("4,12", 16).dec(.06).speed(perlin.range(.2,.6)).delay(.1).room(.1)

$hh: s("[- hh:6!3]*4, [- - - oh:4?]*2").speed(rand.range(.7,.8)).cut(1)
  .delay(.4).velocity(sine.range(.7,1).fast(4)).dist("2:.5").late(.005).gain("<0.9@3 0>/8")

$whistle: note("<a3*3 a4*2>").s("<gm_whistle ->/6").orbit(5).gain(.6)

$blip: note("a2?").s("pulse").att(0).dec(.2).fm("<2 5 8>")
  .delay(.7).late(.05).dist("2:.3").room(.2)

$bass: note("[~ a3@2 a4 e4@4 ~ [d#4!2 - d#4!2 -]@3 ~ [- d#4!3]@2 d4]/2")
  .trans(-24).s("<- sine@4>/4").dec(.8).dist("1:.7")

// Original vibes line from the challenge
$vibes:
  note("<a3 d5> [<a3 a2>,<c4 a3>] [a3,c4,e4] [a3,c4,e4,a4]@13/2")
    .trans(-12)
    .s("<- gm_vibraphone>/8")
    .gain(.3)
    .dec(4.0)
    .late(.02)
    .delay(.2)
    .vib(6)
    .vibmod("<0.5 0>")
    .room(.3)
    .dist("2:.3")

$sshh: note("- - - - - - - a3@2 -").s("gm_applause").att(.50).room(.1).gain(.2).late(.0625)


/////////////////////////////////////////////////
// LIQUID‑STYLE MELODY INSPIRED BY VIBES,
// VOICED TO FIT Dmaj7–Bm7–F#m7–C#m7 CHORDS
/////////////////////////////////////////////////

// This takes the *idea* of the vibes line:
//  - short lead‑in interval
//  - then small two‑note and triad shapes
//  - final longer 4‑note voicing
// but re‑voices everything around your liquid chord progression:
//   Dmaj7 (d, fs, a, c)
//   Bm7   (b, d, fs, a)
//   F#m7  (fs, a, c, e)
//   C#m7  (cs, e, gs, b)

$liquid_vibes:
  note(
    // bar 1 – Dmaj7 colour
    "<d4 a4> [<d4 fs4>,<fs4 a4>] [d4,fs4,a4] [d4,fs4,a4,c5]@13/2 " +
    // bar 2 – Bm7 answer
    "<b3 fs4> [<b3 d4>,<d4 fs4>] [b3,d4,fs4] [b3,d4,fs4,a4]@13/2 " +
    // bar 3 – F#m7 lift
    "<fs4 c5> [<fs4 a4>,<a4 c5>] [fs4,a4,c5] [fs4,a4,c5,e5]@13/2 " +
    // bar 4 – C#m7 resolution
    "<cs4 gs4> [<cs4 e4>,<e4 gs4>] [cs4,e4,gs4] [cs4,e4,gs4,b4]@13/2"
  )
    .s("<- gm_vibraphone>/8")
    .gain(0.32)
    .dec(4.0)
    .late(.02)
    .delay(.22)
    .vib(5.5)
    .vibmod("<0.4 0>")
    .room(.35)
    .dist("2:.25")
